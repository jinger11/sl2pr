import numpy as np
import matplotlib.pyplot as plt

# AND dataset
X = np.array([[0,0],
              [0,1],
              [1,0],
              [1,1]])

y = np.array([0,0,0,1])

# Initialize weights and bias
w = np.zeros(2)
b = 0
learning_rate = 0.1
epochs = 10

# Training using Perceptron Learning Law
for _ in range(epochs):
    for i in range(len(X)):
        linear_output = np.dot(X[i], w) + b
        prediction = 1 if linear_output >= 0 else 0
        error = y[i] - prediction

        # Update rule
        w = w + learning_rate * error * X[i]
        b = b + learning_rate * error

print("Final Weights:", w)
print("Final Bias:", b)

# -----------------------------
# Plotting Decision Boundary
# -----------------------------

plt.figure()

# Plot points
for i in range(len(X)):
    if y[i] == 0:
        plt.scatter(X[i][0], X[i][1], color='red', marker='o')
    else:
        plt.scatter(X[i][0], X[i][1], color='blue', marker='x')

# Decision boundary line (SAFE FIXED LINE)
x_values = np.linspace(-0.5, 1.5, 100)
y_values = -(w[0] * x_values + b) / (w[1] + 1e-6)

plt.plot(x_values, y_values, label="Decision Boundary")

plt.xlim(-0.5, 1.5)
plt.ylim(-0.5, 1.5)
plt.xlabel("Input x1")
plt.ylabel("Input x2")
plt.title("Perceptron Decision Boundary")
plt.legend()
plt.grid(True)
plt.show()
